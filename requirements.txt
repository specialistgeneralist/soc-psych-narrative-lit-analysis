# Portable installation requirements; recommended interpreter: Python 3.11.
# Keep these compatibility rules when updating dependencies from /src imports.
# LLVM is bundled in llvmlite wheels; fail early rather than build it locally.
# Require PyArrow/PyTorch wheels too; do not accidentally build Arrow or Torch.
# Other packages (e.g. annoy) may still need a C/C++ compiler.
--only-binary=numba,llvmlite,pyarrow,torch

# Intel macOS wheels are unavailable for the newer locked Numba/llvmlite pair.
# Markers use the running interpreter's architecture, including under Rosetta.
# The extra NumPy requirement intersects with the general numpy entry below.
# Keep NumPy 1.x for Intel PyTorch 2.2.2, imported transitively by TriMAP.
numba==0.61.2 ; sys_platform == "darwin" and platform_machine == "x86_64"
llvmlite==0.44.0 ; sys_platform == "darwin" and platform_machine == "x86_64"
numpy>=1.24,<2 ; sys_platform == "darwin" and platform_machine == "x86_64"

# PyArrow 26+ requires NumPy 2; keep the non-Intel stack on that ABI.
numpy>=2 ; sys_platform != "darwin" or platform_machine != "x86_64"
pandas>=2.2.2

python-dotenv
PyYAML

openai
anthropic

torch==2.2.2 ; sys_platform == "darwin" and platform_machine == "x86_64"
# Exclude older NumPy-1-only wheels on platforms where NumPy 2 is allowed.
torch>=2.4.1 ; sys_platform != "darwin" or platform_machine != "x86_64"

pyarrow<26 ; sys_platform == "darwin" and platform_machine == "x86_64"
